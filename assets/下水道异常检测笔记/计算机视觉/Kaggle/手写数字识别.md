class DigitDataset(Dataset):
    def __init__(self, images, labels=None):
        self.images = images
        self.labels = labels

    def __len__(self):
        return len(self.images)

    def __getitem__(self, idx):
        # 取出一张图片，原始形状是 [784]
        image = self.images[idx]

        # 归一化：0~255 转成 0~1
        image = image.astype(np.float32) / 255.0

        # 还原成 [1, 28, 28]
        image = image.reshape(1, 28, 28)

        image = torch.tensor(image, dtype=torch.float32)

        if self.labels is not None:
            label = int(self.labels[idx])
            label = torch.tensor(label, dtype=torch.long)
            return image, label

        return image

10. 定义 CNN 模型
class DigitCNN(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()

        self.features = nn.Sequential(
            # 输入: [B, 1, 28, 28]
            nn.Conv2d(1, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),

            nn.Conv2d(32, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),

            nn.MaxPool2d(kernel_size=2),
            nn.Dropout(0.25),

            # 此时: [B, 32, 14, 14]
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),

            nn.Conv2d(64, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),

            nn.MaxPool2d(kernel_size=2),
            nn.Dropout(0.25),

            # 此时: [B, 64, 7, 7]
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True),

            nn.MaxPool2d(kernel_size=2),
            nn.Dropout(0.25),

            # 此时大约: [B, 128, 3, 3]
        )

        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 3 * 3, 256),
            nn.ReLU(inplace=True),
            nn.Dropout(0.5),
            nn.Linear(256, num_classes)
        )

    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return x

12. 训练一个 epoch
def train_one_epoch(model, train_loader, criterion, optimizer, device):
    model.train()

    total_loss = 0.0
    all_preds = []
    all_labels = []

    for images, labels in train_loader:
        images = images.to(device)
        labels = labels.to(device)

        outputs = model(images)
        loss = criterion(outputs, labels)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        total_loss += loss.item() * images.size(0)

        preds = torch.argmax(outputs, dim=1)
        all_preds.extend(preds.detach().cpu().numpy())
        all_labels.extend(labels.detach().cpu().numpy())

    avg_loss = total_loss / len(train_loader.dataset)
    acc = accuracy_score(all_labels, all_preds)

    return avg_loss, acc
13. 验证模型
def validate(model, valid_loader, criterion, device):
    model.eval()

    total_loss = 0.0
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for images, labels in valid_loader:
            images = images.to(device)
            labels = labels.to(device)

            outputs = model(images)
            loss = criterion(outputs, labels)

            total_loss += loss.item() * images.size(0)

            preds = torch.argmax(outputs, dim=1)
            all_preds.extend(preds.detach().cpu().numpy())
            all_labels.extend(labels.detach().cpu().numpy())

    avg_loss = total_loss / len(valid_loader.dataset)
    acc = accuracy_score(all_labels, all_preds)

    return avg_loss, acc

验证阶段要使用：

model.eval()
torch.no_grad()

这样可以关闭 Dropout 的随机性，并且不计算梯度，节省显存和计算。

## 二、手写数字识别实现流程笔记

## 二、对这个项目的理解整理

### 1. 项目任务

Kaggle Digit Recognizer 是一个手写数字识别比赛。

任务目标是：

输入一张 28×28 的灰度手写数字图片
输出该图片对应的数字类别 0~9

这是一个典型的：

图像分类任务

输入是图像，输出是类别。

### 2. 数据格式

训练数据 train.csv 中，每一行表示一张图片。

格式是：

label, pixel0, pixel1, pixel2, ..., pixel783

其中：

label：真实类别，取值 0~9
pixel0 ~ pixel783：图像像素值

每张图片原本是：

28 × 28

总共有：

28 × 28 = 784

个像素。

所以 CSV 里每张图片被展开成了 784 个数字。

### 3. 为什么要归一化？

原始像素范围是：

0 ~ 255

训练时将它除以 255：

image = image / 255.0

变成：

0 ~ 1

这样做的好处是：

让输入数值范围更小
训练更稳定
梯度更新更平滑
模型更容易收敛

如果不归一化，输入像素值比较大，可能导致训练不稳定。

### 4. 为什么要把 784 维向量还原成 1×28×28？

对于普通全连接网络来说，可以直接输入：

[784]

但是 CNN 需要图像的空间结构。

所以将每张图片从：

[784]

还原成：

[1, 28, 28]

其中：

1：通道数，因为是灰度图
28：高度
28：宽度

PyTorch 中 CNN 的输入格式是：

[batch_size, channels, height, width]

所以一个 batch 的输入形状是：

[B, 1, 28, 28]
### 5. 为什么使用 CNN？

手写数字图片有明显的局部结构：

笔画
边缘
弯曲
横线
竖线
圆圈
局部形状

CNN 的卷积核可以学习这些局部模式，这也是在这个项目里选择 CNN 而不是普通全连接网络的原因。

相比全连接网络，CNN 的优势是：

保留图像二维结构
局部连接减少参数量
卷积核共享提高泛化能力
池化增强平移鲁棒性
更适合图像识别任务

对于手写数字识别，CNN 通常比普通 MLP 效果更好。

### 6. CNN 模型结构

本代码中的模型结构大致是：

输入: [B, 1, 28, 28]

Conv2d 1→32
BatchNorm
ReLU

Conv2d 32→32
BatchNorm
ReLU

MaxPool
Dropout

Conv2d 32→64
BatchNorm
ReLU

Conv2d 64→64
BatchNorm
ReLU

MaxPool
Dropout

Conv2d 64→128
BatchNorm
ReLU

MaxPool
Dropout

Flatten

Linear 1152→256
ReLU
Dropout

Linear 256→10

最终输出：

[B, 10]

表示每张图片对应 10 个数字类别的分数。

### 7. 卷积层的作用

卷积层：

nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1)

作用是提取局部特征。

例如第一层卷积可以学习：

边缘
短线条
局部笔画
角点

更深层卷积可以学习：

数字的整体局部结构
圆形结构
交叉结构
复杂笔画组合

在本模型中，卷积层通道数逐渐增加：

1 → 32 → 64 → 128

表示模型从简单特征逐渐学习更复杂的特征。

### 8. ReLU 的作用

ReLU 激活函数：

nn.ReLU(inplace=True)

作用是引入非线性。

如果没有 ReLU，多层卷积和线性层叠加起来仍然近似线性变换，模型表达能力有限。

ReLU 公式是：

ReLU(x) = max(0, x)

它可以让模型学习更复杂的非线性特征。

### 9. BatchNorm 的作用

BatchNorm：

nn.BatchNorm2d(32)

作用是对中间特征进行归一化。

它的好处是：

加快训练收敛
缓解特征分布变化
让训练更稳定
可以使用相对较大的学习率

在 CNN 中，BatchNorm 通常放在：

Conv2d 后面
ReLU 前面

即：

Conv → BatchNorm → ReLU
### 10. MaxPool 的作用

最大池化：

nn.MaxPool2d(kernel_size=2)

作用是降低特征图尺寸。

例如：

28×28 → 14×14
14×14 → 7×7
7×7 → 3×3

它的作用包括：

减少计算量
扩大感受野
保留重要特征
增强一定平移鲁棒性

对于手写数字来说，数字稍微偏移一点，池化可以让模型更加稳定。

### 11. Dropout 的作用

Dropout：

nn.Dropout(0.25)
nn.Dropout(0.5)

训练时会随机丢弃一部分神经元输出。

作用是：

防止过拟合
提高泛化能力
让模型不要过度依赖某些特征

在训练阶段 Dropout 生效，在验证和预测阶段会自动关闭。

这就是为什么验证时要调用：

model.eval()
### 12. Flatten 的作用

卷积层输出是四维张量：

[B, C, H, W]

例如最后可能是：

[B, 128, 3, 3]

全连接层需要二维输入：

[B, features]

所以要使用：

nn.Flatten()

把它变成：

[B, 128×3×3]

也就是：

[B, 1152]
### 13. 全连接层的作用

全连接层负责把 CNN 提取到的图像特征映射成类别分数。

nn.Linear(128 * 3 * 3, 256)
nn.Linear(256, 10)

最后输出 10 个数字类别的 logits：

0 的分数
1 的分数
2 的分数
...
9 的分数

分数最高的类别就是模型预测结果。

### 14. 为什么最后不加 Softmax？

模型最后输出的是 logits，没有加 Softmax。

原因是损失函数使用的是：

nn.CrossEntropyLoss()

它内部已经包含：

LogSoftmax + NLLLoss

所以不需要在模型最后手动加 Softmax，否则反而可能造成数值不稳定。

训练时直接输出 logits 即可。

预测时使用：

torch.argmax(outputs, dim=1)

取分数最大的类别。

### 15. 损失函数 CrossEntropyLoss

手写数字识别是多分类任务，所以使用：

criterion = nn.CrossEntropyLoss()

它要求：

模型输出 shape: [B, 10]
标签 shape: [B]
标签类型: torch.long
标签值: 0~9

如果标签不是 Long 类型，可能会报错。

### 16. 优化器 AdamW

本代码使用：

optimizer = torch.optim.AdamW(...)

AdamW 是 Adam 的改进版本，加入了更合理的权重衰减。

它的作用是：

根据梯度更新模型参数
让 loss 不断下降
让模型预测越来越准确

相比普通 SGD，AdamW 通常更容易训练，收敛速度也比较快。

### 17. 学习率调度器

代码中使用：

CosineAnnealingLR

它会让学习率随着训练逐渐按余弦曲线下降。

作用是：

训练前期学习率较大，快速下降 loss
训练后期学习率较小，精细调整参数

这通常能让模型训练更稳定。

### 18. 训练流程

每个 batch 的训练流程是：

取出一批图片和标签
↓
图片送入模型
↓
模型输出 10 个类别分数
↓
用 CrossEntropyLoss 计算损失
↓
optimizer.zero_grad 清空旧梯度
↓
loss.backward 反向传播计算梯度
↓
optimizer.step 更新模型参数

对应代码：

outputs = model(images)
loss = criterion(outputs, labels)

optimizer.zero_grad()
loss.backward()
optimizer.step()

这是 PyTorch 训练模型的标准流程。

### 19. 验证流程

验证流程和训练类似，但不更新参数。

关闭训练模式
↓
关闭梯度计算
↓
输入验证集图片
↓
得到预测结果

计算 loss 和 accuracy

对应代码：

model.eval()

验证集的作用是：

评估模型泛化能力
判断模型是否过拟合
选择最好的模型
### 20. Accuracy 怎么计算？

模型输出：

[B, 10]

每一行表示一张图片的 10 个类别分数。

用：

preds = torch.argmax(outputs, dim=1)

取得分最高的类别作为预测结果。

然后和真实标签比较：

preds == labels

计算正确比例，就是准确率。

### 21. 保存最佳模型

训练过程中，如果当前验证集准确率更高，就保存模型：

if valid_acc > best_acc:
    best_acc = valid_acc
    torch.save(model.state_dict(), best_model_path)

这样可以避免最后一个 epoch 过拟合。

最终预测时加载验证集表现最好的模型：

model.load_state_dict(torch.load(best_model_path))

## 三、项目总结笔记

项目总结如下：

本项目基于 Kaggle Digit Recognizer 数据集实现手写数字识别。首先读取 CSV 格式图像数据，将每张 28×28 灰度图从 784 维像素向量还原为 1×28×28 的图像张量，并将像素值归一化到 0~1。随后使用 PyTorch 构建 CNN 模型，通过多层卷积提取数字笔画、边缘和局部形状特征，通过池化降低特征图尺寸并增强平移鲁棒性，最后使用全连接层输出 0~9 共 10 个类别的 logits。训练阶段采用 CrossEntropyLoss 作为多分类损失函数，使用 AdamW 优化器和余弦学习率调度器更新模型参数，并在验证集上保存准确率最高的模型。最后使用最佳模型对 Kaggle 测试集进行预测，生成符合比赛格式的 submission.csv 文件。

总结：

这个项目让将图像分类的基础流程完整跑了一遍。从任务角度看，最有价值的不是单独写出 CNN，而是理解数据从 CSV 到 Tensor、从 logits 到预测类别、从验证集指标到最终提交文件的完整链路。后面如果换成更复杂的数据集，也会沿用这个流程：先检查数据，再写 Dataset，最后用一个稳定的 baseline 跑通训练和提交。
